import streamlit as st
import time

# إعداد صفحة التطبيق
st.set_page_config(page_title="مدرب الإلقاء الارتجالي", page_icon="🎤", layout="centered")

st.title("🎤 مدرب الإلقاء والارتجال التفاعلي")
st.write("مرحباً بك! هذا التطبيق يختبر قدرتك على التفكير السريع والارتجال وفق معايير Toastmasters وTED Talks.")

# قائمة بالمواضيع الارتجالية
TOPICS = [
    "لو أُتيحت لك الفرصة لتغيير عادة واحدة يمارسها أغلب الناس يومياً، ما هي هذه العادة؟ ولماذا؟",
    "ما هي أهم مهارة يحتاجها قائد المستقبل في ظل الذكاء الاصطناعي؟",
    "إذا كان بإمكانك إرسال رسالة نصية قصيرة إلى جميع سكان العالم في نفس اللحظة، ماذا ستكتب؟",
    "هل الشغف أم الانضباط هو السر الحقيقي للنجاح؟ وضح وجهة نظرك.",
    "تخيل أنك تُلقي كلمة وداع في مكان عملك الحالي، ما هو الدرس الأهم الذي ستشاركه معهم؟"
]

# إدارة حالة الجلسة (Session State)
if "topic_index" not in st.session_state:
    st.session_state.topic_index = 0

current_topic = TOPICS[st.session_state.topic_index]

st.markdown("---")
st.subheader(f"🎯 التحدي رقم ({st.session_state.topic_index + 1}):")
st.info(f"**\"{current_topic}\"**")

# نصائح سريعة
with st.expander("💡 هيكل الإجابة المثالي (إرشاد للمتدرب)"):
    st.write("""
    - **المقدمة (Hook):** ابدأ بجملة قوية، سؤال، أو حقيقة صادمة.
    - **المضمون (Body):** قدم فكرة رئيسية مع مثال أو قصة قصيرة.
    - **الخاتمة (Call to Action):** اختم برسالة ملهمة أو دعوة للتغيير.
    """)

# نموذج إدخال نص الخطبة
user_speech = st.text_area("اكتب إجابتك الارتجالية هنا (كأنك تتحدث أمام الجمهور):", height=180)

if st.button("إرسال للتقييم 🚀"):
    if len(user_speech.strip()) < 30:
        st.warning("⚠️ الإجابة قصيرة جداً! حاول التوسع في أفكارك واستخدام هيكل الخطبة.")
    else:
        st.success("تم استلام إجابتك بنجاح! جاري تحليل الأداء...")
        time.sleep(1)
        
        # عرض تقييم مبسط (يمكن ربطه بـ AI API للحصول على تقييم ديناميكي)
        st.markdown("### 📊 تقييم الأداء (Sample Rubric)")
        
        st.table({
            "المعيار": ["الهيكل والتنظيم (Structure)", "المحتوى والقصة (Content)", "الإقناع واللغة (Persuasion)"],
            "الدرجة (/10)": [8, 7, 8],
            "نقاط القوة": ["بدأت بمقدمة واضحة", "الفكرة الرئيسية ملموسة", "استخدام مفردات جيدة"]
        })
        
        st.markdown("### 🔑 نقاط التطور الرئيسية (Key Improvement Areas)")
        st.write("1. حاول تعميق الخاتمة بربطها بالمقدمة لتكوين دائرة اكتكال (Full Circle Narrative).")
        st.write("2. أضف مثالاً واقعياً صغيراً لتأكيد الفكرة.")
        
        st.markdown("### 💡 تلميح تحسيني (Refinement Hint)")
        st.info("جرّب استخدام أسلوب 'التباين': قارن بين الوضع الحالي (المشكلة) والوضع المستقبلي (الحل بعد تغيير العادة).")

# زر الانتقال للتحدي التالي
if st.button("التحدي التالي ⏭️"):
    st.session_state.topic_index = (st.session_state.topic_index + 1) % len(TOPICS)
    st.rerun()
